### Python

```python
def load_order(payload: dict) -> dict:
    # Implements the "Orders database" node.
    if not payload.get("id"):
        raise ValueError("id is required")
    return {"id": payload["id"], "status": "done"}
```

### TypeScript

```ts
interface Payload {
  id: string;
}

// Implements the "Orders database" node.
export function loadOrder(payload: Payload): { id: string; status: "done" } {
  if (!payload.id) throw new Error("id is required");
  return { id: payload.id, status: "done" };
}
```
