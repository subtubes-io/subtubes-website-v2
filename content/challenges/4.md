def check_policy(payload: dict) -> dict:
    # Implements the "Check policy" node.

        raise ValueError("id is required")
    return {"id": payload["id"], "status": "done"}

### TypeScript

```ts
interface Payload {
  id: string;
}

// Implements the "Check policy" node.
export function checkPolicy(payload: Payload): { id: string; status: "done" } {
  if (!payload.id) throw new Error("id is required");
  return { id: payload.id, status: "done" };
}
```
